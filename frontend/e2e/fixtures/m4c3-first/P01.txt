<script>
(function () {
  const canvas = document.getElementById('cv');
  const ctx = canvas.getContext('2d');
  const btnPlay = document.getElementById('btnPlay');
  const btnPause = document.getElementById('btnPause');
  const btn0 = document.getElementById('btn0');
  const btnHalf = document.getElementById('btnHalf');
  const btnPi = document.getElementById('btnPi');
  const outPhi = document.getElementById('outPhi');
  const outX = document.getElementById('outX');
  const outSin = document.getElementById('outSin');

  const AMP = 2;
  let phi = 0;
  let playing = false;
  let rafId = null;
  let lastTime = null;
  const SPEED = 1.2; // rad/s

  // 布局参数（逻辑坐标，canvas 会按 CSS 宽度缩放）
  const W = 800, H = 320;
  const marginL = 70, marginR = 70;
  const axisY = 210;
  const plotW = W - marginL - marginR;
  const scale = plotW / (2 * AMP); // 每单位 x 对应像素

  function xToPx(x) {
    return marginL + (x + AMP) * scale;
  }

  function draw() {
    ctx.clearRect(0, 0, W, H);

    const x = AMP * Math.sin(phi);

    // 背景
    ctx.fillStyle = '#fafbff';
    ctx.fillRect(0, 0, W, H);

    // 刻度轴
    ctx.strokeStyle = '#8892b0';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(marginL - 20, axisY);
    ctx.lineTo(W - marginR + 20, axisY);
    ctx.stroke();

    // 箭头
    ctx.fillStyle = '#8892b0';
    ctx.beginPath();
    ctx.moveTo(W - marginR + 20, axisY);
    ctx.lineTo(W - marginR + 8, axisY - 5);
    ctx.lineTo(W - marginR + 8, axisY + 5);
    ctx.closePath();
    ctx.fill();

    // 刻度
    ctx.font = '16px system-ui, sans-serif';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'top';
    ctx.fillStyle = '#334';
    const ticks = [-2, -1, 0, 1, 2];
    ticks.forEach(t => {
      const px = xToPx(t);
      ctx.strokeStyle = '#b8c2d8';
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(px, axisY - 6);
      ctx.lineTo(px, axisY + 6);
      ctx.stroke();
      ctx.fillStyle = '#334';
      ctx.fillText(t.toString(), px, axisY + 12);
    });

    // 平衡位置虚线
    ctx.strokeStyle = '#c5cee0';
    ctx.lineWidth = 1;
    ctx.setLineDash([5, 5]);
    ctx.beginPath();
    ctx.moveTo(xToPx(0), axisY - 70);
    ctx.lineTo(xToPx(0), axisY + 40);
    ctx.stroke();
    ctx.setLineDash([]);

    // 振幅范围标注
    ctx.fillStyle = '#8a94ad';
    ctx.font = '14px system-ui, sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('-A', xToPx(-AMP), axisY + 40);
    ctx.fillText('+A', xToPx(AMP), axisY + 40);

    // 质点
    const px = xToPx(x);
    const py = axisY - 55;
    ctx.beginPath();
    ctx.arc(px, py, 14, 0, Math.PI * 2);
    ctx.fillStyle = '#3b5bdb';
    ctx.fill();
    ctx.strokeStyle = '#fff';
    ctx.lineWidth = 3;
    ctx.stroke();

    // 位置投影线
    ctx.strokeStyle = '#93b4ff';
    ctx.lineWidth = 2;
    ctx.setLineDash([4, 4]);
    ctx.beginPath();
    ctx.moveTo(px, py + 14);
    ctx.lineTo(px, axisY);
    ctx.stroke();
    ctx.setLineDash([]);

    // 位置数值
    ctx.fillStyle = '#3b5bdb';
    ctx.font = 'bold 16px system-ui, sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('x = ' + x.toFixed(2), px, py - 34);

    // 相位弧示意（左上角）
    const cx = 90, cy = 70, r = 40;
    ctx.strokeStyle = '#dde3f0';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.arc(cx, cy, r, 0, Math.PI * 2);
    ctx.stroke();

    // 角度弧
    const startAngle = -Math.PI / 2;
    const endAngle = startAngle + phi;
    ctx.strokeStyle = '#3b5bdb';
    ctx.lineWidth = 3;
    ctx.beginPath();
    if (phi >= 0) {
      ctx.arc(cx, cy, r, startAngle, endAngle, false);
    } else {
      ctx.arc(cx, cy, r, startAngle, endAngle, true);
    }
    ctx.stroke();

    // 半径线
    ctx.strokeStyle = '#3b5bdb';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(cx, cy);
    ctx.lineTo(cx + r * Math.cos(endAngle), cy + r * Math.sin(endAngle));
    ctx.stroke();

    // 圆心
    ctx.beginPath();
    ctx.arc(cx, cy, 3, 0, Math.PI * 2);
    ctx.fillStyle = '#3b5bdb';
    ctx.fill();

    // 标签
    ctx.fillStyle = '#555';
    ctx.font = '14px system-ui, sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('φ', cx, cy + r + 22);

    // 更新读数
    outPhi.textContent = phi.toFixed(2) + ' rad';
    outX.textContent = x.toFixed(2);
    outSin.textContent = Math.sin(phi).toFixed(3);

    // 更新按钮高亮
    btn0.classList.toggle('active', Math.abs(phi) < 0.01);
    btnHalf.classList.toggle('active', Math.abs(phi - Math.PI / 2) < 0.01);
    btnPi.classList.toggle('active', Math.abs(phi - Math.PI) < 0.01);

    // 通知 studyUI
    if (window.studyUI && window.studyUI.setState) {
      window.studyUI.setState({
        phi: parseFloat(phi.toFixed(3)),
        x: parseFloat(x.toFixed(3)),
        sinPhi: parseFloat(Math.sin(phi).toFixed(3)),
        playing: playing
      });
    }
  }

  function loop(timestamp) {
    if (!playing) return;
    if (lastTime === null) lastTime = timestamp;
    const dt = (timestamp - lastTime) / 1000;
    lastTime = timestamp;
    phi += SPEED * dt;
    // 保持在 [0, 2π) 循环更自然，但允许持续增长也可以；这里取模便于读数
    if (phi > 2 * Math.PI) phi -= 2 * Math.PI;
    draw();
    rafId = requestAnimationFrame(loop);
  }

  function setPlaying(v) {
    playing = v;
    btnPlay.setAttribute('aria-pressed', v ? 'true' : 'false');
    btnPause.setAttribute('aria-pressed', v ? 'true' : 'false');
    if (v) {
      lastTime = null;
      rafId = requestAnimationFrame(loop);
    } else {
      if (rafId !== null) {
        cancelAnimationFrame(rafId);
        rafId = null;
      }
    }
  }

  function setPhi(v) {
    phi = v;
    draw();
  }

  btnPlay.addEventListener('click', () => setPlaying(true));
  btnPause.addEventListener('click', () => setPlaying(false));
  btn0.addEventListener('click', () => { setPlaying(false); setPhi(0); });
  btnHalf.addEventListener('click', () => { setPlaying(false); setPhi(Math.PI / 2); });
  btnPi.addEventListener('click', () => { setPlaying(false); setPhi(Math.PI); });

  // 尊重减少动画偏好
  if (window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
    setPlaying(false);
  }

  // 恢复状态支持
  if (window.studyUI && window.studyUI.onRestore) {
    window.studyUI.onRestore(function (snap) {
      if (snap && typeof snap.phi === 'number') {
        phi = snap.phi;
      }
      if (snap && typeof snap.playing === 'boolean') {
        setPlaying(snap.playing);
      }
      draw();
    });
  }

  draw();
})();
</script>
</body>

### 玩法与视觉设计

页面围绕“相位驱动位置”这一核心关系，把抽象的三角函数变成看得见的运动。

- **画布与刻度**：水平轴上标出 -2、-1、0、1、2 五个位置刻度，虚线标出平衡位置，让质点位置一目了然。
- **相位控制**：三个快捷按钮分别将 φ 设为 0、π/2、π，点击后质点立即跳到对应位置，同时按钮高亮显示当前状态。
- **播放与暂停**：播放时相位匀速增加，质点来回运动；暂停后画面定格，方便观察某一时刻的数值。
- **实时读数**：下方同步显示 φ、x 和 sin(φ) 的精确数值，左上角还有一个小圆环直观展示相位角的大小。
- **响应式适配**：画布宽度随容器缩放，按钮和字号满足触控要求，手机上不会横向溢出。
---

**优化建议：** 您可以在代码开头的 `SPEED = 1.2` 处调整播放速度（数值越大越快），在 `AMP = 2` 处修改振幅，刻度数组 `ticks` 也会随之适配。
