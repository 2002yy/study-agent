<script>
  (function() {
    // ---------- 坐标映射 ----------
    // 可视区域：x 从 -10 到 10 映射到 100 到 300（每单位10px）；y 从 -10 到 10 映射到 230 到 30（每单位10px）
    const SVG_W = 400, SVG_H = 260;
    const ORIGIN_X = 200, ORIGIN_Y = 130;
    const SCALE = 10; // 每单位10像素

    function toSvgX(x) { return ORIGIN_X + x * SCALE; }
    function toSvgY(y) { return ORIGIN_Y - y * SCALE; }

    // ---------- 获取 SVG 分组 ----------
    const svg = document.querySelector('svg');
    const gridLayer = document.getElementById('grid-layer');
    const axisLayer = document.getElementById('axis-layer');
    const graphLayer = document.getElementById('graph-layer');
    const pointLayer = document.getElementById('point-layer');

    // ---------- 清空图层 ----------
    function clearLayer(layer) { while (layer.firstChild) layer.removeChild(layer.firstChild); }

    // ---------- 绘制网格 ----------
    function drawGrid() {
      clearLayer(gridLayer);
      const step = 2; // 每2单位一条网格线
      // 垂直网格线（x = -10, -8, ..., 10）
      for (let x = -10; x <= 10; x += step) {
        if (x === 0) continue; // 轴线单独画
        const sx = toSvgX(x);
        const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
        line.setAttribute('x1', sx);
        line.setAttribute('y1', toSvgY(-10));
        line.setAttribute('x2', sx);
        line.setAttribute('y2', toSvgY(10));
        line.setAttribute('class', 'grid-line');
        gridLayer.appendChild(line);
      }
      // 水平网格线（y = -10, -8, ..., 10）
      for (let y = -10; y <= 10; y += step) {
        if (y === 0) continue;
        const sy = toSvgY(y);
        const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
        line.setAttribute('x1', toSvgX(-10));
        line.setAttribute('y1', sy);
        line.setAttribute('x2', toSvgX(10));
        line.setAttribute('y2', sy);
        line.setAttribute('class', 'grid-line');
        gridLayer.appendChild(line);
      }
    }

    // ---------- 绘制坐标轴和刻度 ----------
    function drawAxes() {
      clearLayer(axisLayer);
      // X 轴
      const xAxis = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      xAxis.setAttribute('x1', toSvgX(-10));
      xAxis.setAttribute('y1', toSvgY(0));
      xAxis.setAttribute('x2', toSvgX(10));
      xAxis.setAttribute('y2', toSvgY(0));
      xAxis.setAttribute('class', 'axis-line');
      axisLayer.appendChild(xAxis);
      // Y 轴
      const yAxis = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      yAxis.setAttribute('x1', toSvgX(0));
      yAxis.setAttribute('y1', toSvgY(-10));
      yAxis.setAttribute('x2', toSvgX(0));
      yAxis.setAttribute('y2', toSvgY(10));
      yAxis.setAttribute('class', 'axis-line');
      axisLayer.appendChild(yAxis);

      // 刻度标签（只标几个关键数字，避免拥挤）
      const tickValues = [-5, 0, 5];
      tickValues.forEach(val => {
        if (val === 0) return;
        // X 轴刻度
        const tx = toSvgX(val);
        const ty = toSvgY(0);
        const labelX = document.createElementNS('http://www.w3.org/2000/svg', 'text');
        labelX.setAttribute('x', tx);
        labelX.setAttribute('y', ty + 18);
        labelX.setAttribute('text-anchor', 'middle');
        labelX.setAttribute('class', 'axis-label');
        labelX.textContent = val;
        axisLayer.appendChild(labelX);
        // Y 轴刻度
        const labelY = document.createElementNS('http://www.w3.org/2000/svg', 'text');
        labelY.setAttribute('x', toSvgX(0) - 12);
        labelY.setAttribute('y', toSvgY(val) + 5);
        labelY.setAttribute('text-anchor', 'end');
        labelY.setAttribute('class', 'axis-label');
        labelY.textContent = val;
        axisLayer.appendChild(labelY);
      });
      // 原点标签 0
      const zeroLabel = document.createElementNS('http://www.w3.org/2000/svg', 'text');
      zeroLabel.setAttribute('x', toSvgX(0) - 10);
      zeroLabel.setAttribute('y', toSvgY(0) + 18);
      zeroLabel.setAttribute('text-anchor', 'end');
      zeroLabel.setAttribute('class', 'axis-label');
      zeroLabel.textContent = '0';
      axisLayer.appendChild(zeroLabel);
    }

    // ---------- 绘制直线 y = 2x + 1 ----------
    function drawLine() {
      clearLayer(graphLayer);
      // 计算可视区域内的线段端点（x从-6到6，足够覆盖）
      const x1 = -6, y1 = 2 * x1 + 1; // -11
      const x2 = 6, y2 = 2 * x2 + 1;  // 13
      // 但为了不超出SVG太多，使用clipPath。简单起见，我们直接画，因为超出部分会被SVG视口裁剪。
      const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      line.setAttribute('x1', toSvgX(x1));
      line.setAttribute('y1', toSvgY(y1));
      line.setAttribute('x2', toSvgX(x2));
      line.setAttribute('y2', toSvgY(y2));
      line.setAttribute('class', 'line-graph');
      graphLayer.appendChild(line);
    }

    // ---------- 绘制当前点 ----------
    function drawPoint(x) {
      clearLayer(pointLayer);
      const y = 2 * x + 1;
      const cx = toSvgX(x);
      const cy = toSvgY(y);

      // 点
      const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
      circle.setAttribute('cx', cx);
      circle.setAttribute('cy', cy);
      circle.setAttribute('r', 6);
      circle.setAttribute('class', 'point-circle');
      pointLayer.appendChild(circle);

      // 标签背景（白色圆角矩形）
      const labelText = `(${x.toFixed(1)}, ${y.toFixed(1)})`;
      // 先创建文本以测量宽度？但测量需要DOM渲染，在SVG中可以用getComputedTextLength，但需要先挂载。简化：用固定宽度估算，或者直接画文本，背景用矩形宽度动态计算。
      // 为了简单，我们使用带背景的文本，背景宽度根据文本长度估算。
      const textEl = document.createElementNS('http://www.w3.org/2000/svg', 'text');
      textEl.setAttribute('x', cx + 12);
      textEl.setAttribute('y', cy - 10);
      textEl.setAttribute('class', 'point-label');
      textEl.textContent = labelText;
      // 估算宽度：每个字符约8px
      const approxWidth = labelText.length * 8 + 16;
      const rect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
      rect.setAttribute('x', cx + 8);
      rect.setAttribute('y', cy - 26);
      rect.setAttribute('width', approxWidth);
      rect.setAttribute('height', 24);
      rect.setAttribute('class', 'value-badge-bg');
      rect.setAttribute('rx', 8);
      pointLayer.appendChild(rect);
      pointLayer.appendChild(textEl);
    }

    // ---------- 更新所有 ----------
    function updateAll(x) {
      // 更新输出
      const y = 2 * x + 1;
      document.getElementById('xOut').textContent = x.toFixed(1);
      document.getElementById('yOut').textContent = y.toFixed(1);

      // 重绘点
      drawPoint(x);

      // 通知 studyUI 状态
      if (window.studyUI && typeof window.studyUI.setState === 'function') {
        window.studyUI.setState({
          x: parseFloat(x.toFixed(1)),
          y: parseFloat(y.toFixed(1)),
          observation: `当前 x = ${x.toFixed(1)}，y = 2x + 1 = ${y.toFixed(1)}`
        });
      }
    }

    // ---------- 初始化图形（网格、轴、直线） ----------
    drawGrid();
    drawAxes();
    drawLine();

    // ---------- 绑定控件 ----------
    const slider = document.getElementById('xSlider');
    const setXneg1 = document.getElementById('setXneg1');
    const setX3 = document.getElementById('setX3');
    const resetBtn = document.getElementById('resetBtn');

    // 滑块事件
    slider.addEventListener('input', (e) => {
      const x = parseFloat(e.target.value);
      updateAll(x);
    });

    // 按钮事件
    setXneg1.addEventListener('click', () => {
      slider.value = -1;
      updateAll(-1);
    });

    setX3.addEventListener('click', () => {
      slider.value = 3;
      updateAll(3);
    });

    resetBtn.addEventListener('click', () => {
      slider.value = 3;
      updateAll(3);
    });

    // 初始状态 x = 3
    slider.value = 3;
    updateAll(3);

    // ---------- 恢复上次状态（如果支持） ----------
    if (window.studyUI && typeof window.studyUI.onRestore === 'function') {
      window.studyUI.onRestore((snapshot) => {
        // 从快照恢复 x，如果无效则回退到 3
        let restoredX = 3;
        if (snapshot && typeof snapshot.x === 'number' && !isNaN(snapshot.x)) {
          // 限制在滑块范围内
          restoredX = Math.min(5, Math.max(-5, snapshot.x));
        }
        slider.value = restoredX;
        updateAll(restoredX);
      });
    }
  })();
